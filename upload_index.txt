function h(tag, props, ...kids) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(props || {})) {
    if (v == null || v === false) continue;
    if (k === 'class') el.className = v;
    else if (k === 'style' && typeof v === 'object') {
      for (const [sk, sv] of Object.entries(v)) sk.startsWith('--') ? el.style.setProperty(sk, sv) : (el.style[sk] = sv);
    } else if (k.startsWith('on') && typeof v === 'function') el.addEventListener(k.slice(2).toLowerCase(), v);
    else if (k === 'html') el.innerHTML = v;
    else if (k === 'value') el.value = v;
    else el.setAttribute(k, v === true ? '' : v);
  }
  for (const c of kids.flat(Infinity)) {
    if (c == null || c === false) continue;
    el.append(c instanceof Node ? c : document.createTextNode(String(c)));
  }
  return el;
}

async function makeFirebaseBackend(cfg) {
  const appMod = await import(`https://www.gstatic.com/firebasejs/${FB_VER}/firebase-app.js`);
  const D = await import(`https://www.gstatic.com/firebasejs/${FB_VER}/firebase-database.js`);
  const db = D.getDatabase(appMod.initializeApp(cfg));
  const R = p => D.ref(db, p);
  const val = async p => (await D.get(R(p))).val();
  return {
    demo: false,
    async getConfig() { return (await val('config')) || {}; },
    async setTeacherHash(v) { await D.set(R('config/teacherHash'), v); },
    async createTable(code, meta) {
      await D.set(R(`tables/${code}/meta`), meta);
      await D.set(R(`tableIndex/${code}`), { name: meta.name, createdAt: meta.createdAt });
    },
    getTableMeta: code => val(`tables/${code}/meta`),
    async listTables() { const v = (await val('tableIndex')) || {}; return Object.entries(v).map(([code, m]) => ({ code, ...m })); },
    async updateTableMeta(code, patch) {
      await D.update(R(`tables/${code}/meta`), patch);
      if (patch.name) await D.update(R(`tableIndex/${code}`), { name: patch.name });
    },
    async deleteTable(code) { await D.remove(R(`blobs/${code}`)); await D.remove(R(`tables/${code}`)); await D.remove(R(`tableIndex/${code}`)); },
    watchTable(code, cb) { return D.onValue(R(`tables/${code}`), s => cb(s.val()), e => { console.error(e); toast('通信エラー：' + e.message); }); },
    newId: () => D.push(R('_ids')).key,
    async addFile(code, id, meta, chunks, onProgress) {
      for (let i = 0; i < chunks.length; i++) { await D.set(R(`blobs/${code}/${id}/${i}`), chunks[i]); onProgress?.((i + 1) / chunks.length); }
      await D.set(R(`tables/${code}/files/${id}`), meta);
    },
    async getFileData(code, id) {
      const v = await val(`blobs/${code}/${id}`); if (!v) return null;
      return Array.isArray(v) ? v : Object.keys(v).sort((a, b) => a - b).map(k => v[k]);
    },
    updateFile: (code, id, patch) => D.update(R(`tables/${code}/files/${id}`), patch),
    async replaceFile(code, id, oldKey, newKey, patch, chunks, onProgress) {
      for (let i = 0; i < chunks.length; i++) { await D.set(R(`blobs/${code}/${newKey}/${i}`), chunks[i]); onProgress?.((i + 1) / chunks.length); }
      await D.update(R(`tables/${code}/files/${id}`), patch);
      if (oldKey !== newKey) await D.remove(R(`blobs/${code}/${oldKey}`)).catch(() => {});
    },
    tset: (code, path, val) => val == null ? D.remove(R(`tables/${code}/${path}`)) : D.set(R(`tables/${code}/${path}`), val),
    tupdate: (code, path, patch) => D.update(R(`tables/${code}/${path}`), patch),
    addComment: (code, fid, cid, c) => D.set(R(`tables/${code}/comments/${fid}/${cid}`), c),
    deleteComment: (code, fid, cid) => D.remove(R(`tables/${code}/comments/${fid}` + (cid ? `/${cid}` : ''))),
    async deleteFile(code, id, key) {
      await D.remove(R(`tables/${code}/files/${id}`));
      await D.remove(R(`blobs/${code}/${key || id}`));
      if (key && key !== id) await D.remove(R(`blobs/${code}/${id}`)).catch(() => {});
    }
  };
}

/* ---------------- binary <-> base64 chunks ---------------- */
function bytesToChunks(u8, size = 393216) {
  const out = [];
  for (let i = 0; i < u8.length; i += size) {
    const sub = u8.subarray(i, i + size); let s = '';
    for (let j = 0; j < sub.length; j += 0x8000) s += String.fromCharCode.apply(null, sub.subarray(j, j + 0x8000));
    out.push(btoa(s));
  }
  return out;
}
function chunksToBytes(chunks) {
  const parts = chunks.map(b => { const s = atob(b); const u = new Uint8Array(s.length); for (let i = 0; i < s.length; i++) u[i] = s.charCodeAt(i); return u; });
  const out = new Uint8Array(parts.reduce((a, p) => a + p.length, 0)); let o = 0;
  for (const p of parts) { out.set(p, o); o += p.length; }
  return out;
}

function openPdf(bytes) {

  return pdfjsLib.getDocument({ data: bytes.slice(0), cMapUrl: PDFJS + 'cmaps/', cMapPacked: true, standardFontDataUrl: PDFJS + 'standard_fonts/' }).promise;
}
async function makeThumb(pdf) {
  const page = await pdf.getPage(1);
  const vp0 = page.getViewport({ scale: 1 });
  const vp = page.getViewport({ scale: 260 / vp0.width });
  const c = document.createElement('canvas'); c.width = Math.round(vp.width); c.height = Math.round(Math.min(vp.height, 260 * 1.6));
  const ctx = c.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, c.width, c.height);
  await page.render({ canvasContext: ctx, viewport: vp }).promise;
  return c.toDataURL('image/jpeg', 0.7);
}
async function docxToPdf(file) {

async function getPdfBytes(f) {

// 画像 → PDF（複数枚なら1ページずつ）。長辺2400pxまで縮小してJPEGで埋め込む
async function imageToJpeg(file) {
  const url = URL.createObjectURL(file);
  try {
    const img = await new Promise((res, rej) => { const i = new Image(); i.onload = () => res(i); i.onerror = () => rej(new Error(`画像を読み込めませんでした（${file.name}）`)); i.src = url; });
    const sc = Math.min(1, 2400 / Math.max(img.naturalWidth, img.naturalHeight));
    const w = Math.max(1, Math.round(img.naturalWidth * sc)), hh = Math.max(1, Math.round(img.naturalHeight * sc));
    const c = document.createElement('canvas'); c.width = w; c.height = hh;
    const ctx = c.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, w, hh); ctx.drawImage(img, 0, 0, w, hh);
    return { data: c.toDataURL('image/jpeg', 0.88), w, h: hh };
  } finally { URL.revokeObjectURL(url); }
}
async function imagesToPdf(files) {
  if (!window.jspdf?.jsPDF) await loadScript(LIB.jspdf);
  const { jsPDF } = window.jspdf; let doc;
  for (const f of files) {
    const { data, w, h: hh } = await imageToJpeg(f);
    const o = w > hh ? 'l' : 'p';
    if (!doc) doc = new jsPDF({ unit: 'px', format: [w, hh], orientation: o, hotfixes: ['px_scaling'], compress: true });
    else doc.addPage([w, hh], o);
    doc.addImage(data, 'JPEG', 0, 0, w, hh);
  }
  return new Uint8Array(doc.output('arraybuffer'));
}
// アップロード・置き換え共通：PDF化してページ数とサムネイルを作る（files は1つ、または画像の配列）
async function preparePdf(files, onStatus) {
  files = [].concat(files);
  let bytes, srcType;
  if (files.every(isImage)) { onStatus?.('画像をPDFに変換中…'); bytes = await imagesToPdf(files); srcType = 'image'; }
  else if (isDocx(files[0])) { onStatus?.('変換中…'); bytes = await docxToPdf(files[0]); srcType = 'docx'; }
  else { bytes = new Uint8Array(await files[0].arrayBuffer()); srcType = 'pdf'; }
  if (bytes.length > MAX_BYTES) throw new Error(`大きすぎます（${fmtSize(bytes.length)}、上限15MB）`);
  onStatus?.('確認中…');
  const pdf = await openPdf(bytes);
  const pages = pdf.numPages; const thumb = await makeThumb(pdf).catch(() => null); pdf.destroy();
  return { bytes, pages, thumb, srcType };
}

function enterTable(code) {
  stopWatch();
  S.code = code; S.prefs = loadPrefs(code); S.files = []; S.table = null;
  S.ui.select = false; S.ui.selected.clear(); S.ui.filter = 'all'; S.ui.search = '';
  history.replaceState(null, '', '#t=' + code);
  buildTableShell();
  S.unwatch = S.backend.watchTable(code, t => {
    if (S.code !== code) return;
    if (!t || !t.meta) { closeViewer(true); renderJoin('テーブルが見つかりません（削除された可能性があります）'); history.replaceState(null, '', location.pathname); return; }
    const first = !S.table;
    S.table = t.meta;
    const recent = LS.get('yk:recent', []).filter(r => r.code !== code);
    recent.unshift({ code, name: t.meta.name }); LS.set('yk:recent', recent.slice(0, 8));
    const prevComments = S.comments;
    S.comments = t.comments || {};
    const prevPolls = S.polls; S.polls = t.polls || {}; S.votes = t.votes || {};
    S.files = Object.entries(t.files || {}).map(([id, f]) => ({ id, ...f }))
      .sort((a, b) => (a.createdAt - b.createdAt) || (a.id < b.id ? -1 : 1));
    S.files.forEach((f, i) => { f.no = i + 1; });
    const ids = new Set(S.files.map(f => f.id));
    for (const id of [...S.ui.selected]) if (!ids.has(id)) S.ui.selected.delete(id);
    updateTable();
    checkPanesForChanges();
    if (!first) { notifyNewComments(prevComments); notifyNewPolls(prevPolls); }
    S.pollViews.forEach(fn => fn());
    V.panes?.forEach(p => { p.updateLabel(); if (p.cpanel && !p.cpanel.hidden) p.drawComments(); });
    if (V.el && !V.el.hidden) drawTabs();
  });
}

function visibleFiles() {
  let fs = S.files.slice(); const P = S.prefs, F = S.ui.filter;
  const q = S.ui.search.trim().toLowerCase();
  if (q) fs = fs.filter(f => fileLabel(f).toLowerCase().includes(q));
  if (F === 'fav') fs = fs.filter(f => P.fav[f.id]);
  else if (F === 'none') fs = fs.filter(f => !groupOf(f.id));
  else if (F.startsWith('g:')) fs = fs.filter(f => P.assign[f.id] === F.slice(2));
  if (S.ui.sort === 'title') {
    const coll = new Intl.Collator('ja', { numeric: true, sensitivity: 'base' });
    fs.sort((a, b) => coll.compare(fileLabel(a), fileLabel(b)) || a.no - b.no);
  } else if (S.ui.sort === 'random') {
    // 端末ごとに固定の並び（シャッフルを押すまで変わらない。新しい文書もランダムな位置に入る）
    let seed = LS.get('yk:seed:' + S.code, null); if (!seed) { seed = rid(6); LS.set('yk:seed:' + S.code, seed); }
    const hash = s => { let x = 2166136261; for (let i = 0; i < s.length; i++) { x ^= s.charCodeAt(i); x = Math.imul(x, 16777619); } return x >>> 0; };
    const k = new Map(fs.map(f => [f.id, hash(seed + f.id)]));
    fs.sort((a, b) => k.get(a.id) - k.get(b.id));
  }
  return fs;
}

}

async function downloadFile(f) {

    const url = URL.createObjectURL(new Blob([bytes], { type: 'application/pdf' }));

}

async function startUpload(fileList) {
  if (!S.table) return;
  if (S.table.locked && !S.teacher) return toast('現在アップロードは締め切られています');
  const ok = fileList.filter(okFile);
  if (fileList.some(f => /\.doc$/i.test(f.name))) toast('古い形式(.doc)は非対応です。Wordで .docx か PDF に保存し直してください');
  else if (ok.length < fileList.length) toast('PDF・Word(.docx)・画像のみアップできます');
  if (!ok.length) return;
  const imgs = ok.filter(isImage).sort((a, b) => new Intl.Collator('ja', { numeric: true }).compare(a.name, b.name));
  let merge = false;
  let rows = [];
  const listEl = h('div');
  const buildRows = () => {
    const jobs = merge ? [imgs, ...ok.filter(f => !isImage(f)).map(f => [f])] : ok.map(f => [f]);
    rows = jobs.map(files => {
      const f0 = files[0], type = files.every(isImage) ? 'image' : isDocx(f0) ? 'docx' : 'pdf';
      const size = files.reduce((a, f) => a + f.size, 0);
      const r = { files, title: h('input', { class: 'input', placeholder: `タイトル（${files.length > 1 ? '画像' + files.length + '枚' : f0.name}）` }),
        stat: h('span', {}, files.length > 1 ? `画像${files.length}枚 → 1つのPDFに` : type === 'docx' ? 'Word → PDFに変換してアップ' : type === 'image' ? '画像 → PDFに変換してアップ' : '待機中'), bar: h('i') };
      r.el = h('div', { class: 'up-row' }, h('span', { class: 'badge' + TYPE_CLS[type] }, TYPE_LABEL[type]),
        h('div', { class: 'info' }, r.title, h('div', { class: 'stat' }, fmtSize(size), r.stat, h('span', { class: 'bar' }, r.bar))));
      return r;
    });
    listEl.replaceChildren(...rows.map(r => r.el));
    if (titleEl) titleEl.textContent = `アップロード（${rows.length}件）`;
  };
  let titleEl = null;
  buildRows();
  const mergeChk = h('input', { type: 'checkbox', onchange: e => { merge = e.target.checked; buildRows(); } });
  const sigIn = h('input', { class: 'input', value: LS.get('yk:sig', '') || S.me.name, placeholder: '署名（先生の指示どおりに）' });
  const upErr = h('div', { class: 'err' });
  let busy = false, ov = null;
  await modal({ bind: (_c, o) => { ov = o; titleEl = o.querySelector('.modal-h h3'); }, title: `アップロード（${rows.length}件）`, wide: true, body: [
    h('p', { class: 'hint', style: { margin: '0 0 8px' } }, '署名とタイトルは、先生の指示どおりに書いてください。'),
    h('div', { class: 'field' }, h('label', {}, '署名'), sigIn),
    h('div', { style: { fontSize: '13px', fontWeight: '600', margin: '4px 0 0' } }, 'タイトル'),
    listEl,
    imgs.length >= 2 ? h('label', { class: 'merge-opt' }, mergeChk, h('div', {}, h('b', {}, `画像${imgs.length}枚をまとめて1つの文書にする`),
      h('small', {}, '手書きの原稿を何枚も撮ったときなどに。ページの順番はファイル名順です。'))) : null,
    upErr,
    ok.some(isDocx) ? h('p', { class: 'hint' }, 'Wordはこの端末でPDFに変換されます。文字・表・画像は残りますが、縦書きや細かいレイアウトは崩れることがあります。崩れる場合は Word で「PDFとして保存」してからアップしてください。') : null,
    imgs.length ? h('p', { class: 'hint' }, '画像（JPG・PNGなど）はPDFに変換して置かれます。大きな写真は長い辺が2400pxになるよう縮小されます。') : null
  ], actions: [
    { text: 'キャンセル', value: null, onClick: () => busy ? false : undefined },
    { text: 'アップロード', kind: 'primary', onClick: async () => {
      if (busy) return false;
      const sig = sigIn.value.trim();
      if (!sig) { upErr.textContent = '署名を入れてください'; sigIn.focus(); return false; }
      const emptyT = rows.find(r => !r.done && !r.title.value.trim());
      if (emptyT) { upErr.textContent = 'タイトルを入れてください'; emptyT.title.focus(); return false; }
      upErr.textContent = ''; LS.set('yk:sig', sig);
      busy = true;
      ov.dataset.busy = '1';
      ov.querySelectorAll('.modal-f .btn, .up-row input, .merge-opt input, .field input').forEach(b => b.disabled = true);
      let fail = 0;
      for (const r of rows) {
        if (r.done) continue;
        try {
          const { bytes, pages, thumb, srcType } = await preparePdf(r.files, t => { r.stat.textContent = t; });
          r.stat.textContent = 'アップロード中…';
          const id = S.backend.newId();
          const meta = { title: r.title.value.trim(), srcType, version: 1, anonymous: false, author: sig,
            pages, size: bytes.length, thumb, createdAt: Date.now() };
          S.prefs.mine[id] = 1; S.prefs.seen[id] = 1; savePrefs();
          await S.backend.addFile(S.code, id, meta, bytesToChunks(bytes), p => { r.bar.style.width = Math.round(p * 100) + '%'; });
          idb.set('pdf', pdfKey(id), bytes.buffer.slice(0));
          r.bar.style.width = '100%'; r.stat.textContent = '完了 ✓'; r.done = true; r.stat.style.color = '';
        } catch (e) {
          console.error(e); fail++; r.stat.textContent = '失敗：' + (e.message || e); r.stat.style.color = 'var(--danger)';
        }
      }
      delete ov.dataset.busy; busy = false;
      if (!fail) { toast('アップロードしました'); return 'done'; }
      ov.querySelectorAll('.modal-f .btn').forEach(b => b.disabled = false);
      return false;
    } }
  ] });
}

/* =====================================================================
   投票
   polls/{pid}: { title, target:'docs'|'choices', choices:[{id,label}], votes:n,
                  multi, noSelf, change, results:'teacher'|'publish'|'live', published, named, status:'open'|'closed', createdAt }
   votes/{pid}/{voterKey}: { items:{targetId:count}, at, name? }   ※voterKeyは投票ごとのランダム値（端末内に保存）
   ===================================================================== */
const pollList = () => Object.entries(S.polls).map(([id, p]) => ({ id, ...p })).sort((a, b) => b.createdAt - a.createdAt);
const myVoteKey = pid => S.prefs.vkeys[pid];
const myVote = pid => { const k = myVoteKey(pid); return k ? S.votes[pid]?.[k] || null : null; };
const canSeeResults = p => S.teacher || p.results === 'live' || (p.results === 'publish' && p.published);
function pollTargets(p) {
  if (p.target === 'choices') return (p.choices || []).map(c => ({ id: c.id, label: c.label }));
  return S.files.map(f => ({ id: f.id, label: fileLabel(f), thumb: f.thumb, mine: !!S.prefs.mine[f.id] }));
}
function tally(pid) {
  const t = {}, who = {}; let voters = 0;
  for (const v of Object.values(S.votes[pid] || {})) {
    voters++;
    for (const [k, n] of Object.entries(v.items || {})) { t[k] = (t[k] || 0) + n; if (v.name) (who[k] ||= []).push(n > 1 ? `${v.name}(${n})` : v.name); }
  }
  return { t, who, voters };
}

}

function voteDialog(pid) {
  const p0 = S.polls[pid]; if (!p0) return;
  const cur = myVote(pid);

function resultsDialog(pid) {
  const body = h('div');
  const draw = () => {
    const p = S.polls[pid]; if (!p) { body.replaceChildren(h('p', { class: 'hint' }, 'この投票は削除されました。')); return; }

    const rows = pollTargets(p).map(x => ({ ...x, n: t[x.id] || 0 })).sort((a, b) => b.n - a.n);
    const max = Math.max(1, ...rows.map(r => r.n));
    let rank = 0, prev = null;

      h('p', { style: { margin: '0 0 6px', fontWeight: '700', fontSize: '15px' } }, p.title),

        return h('div', { class: 'rrow' }, h('span', { class: 'rk' }, r.n ? rank : '–'),
          h('div', { class: 'rl' }, r.label, r.sub ? h('small', { style: { color: 'var(--muted)', marginLeft: '6px' } }, r.sub) : null),

function openViewer(ids, opt = {}) {
  if (!V.el) buildViewer();

function addPane(id, opt = {}) {
  if (V.panes.length >= 3) return toast('並べられるのは3つまでです');

/* ---------------- コメント ----------------
   self  : 自分用に残す（この端末だけに保存。置き換えで消える）
   author: 書き手に伝える（自分と、その文書をアップした端末にだけ表示）
   all   : みんなに共有（全員に表示） */
function visibleComments(f) {
  const out = [];
  const n = S.prefs.notes[f.id];
  if (n && n.v === verOf(f)) for (const c of n.list) out.push({ ...c, type: 'self' });
  const mineFile = !!S.prefs.mine[f.id];
  for (const [id, c] of Object.entries(S.comments[f.id] || {})) {
    if (c.type === 'all' || (c.type === 'author' && (mineFile || S.prefs.myc[id]))) out.push({ id, ...c });
  }
  return out.sort((a, b) => a.createdAt - b.createdAt);
}
function unreadCount(f, cs) {
  const t = S.prefs.cseen[f.id] || 0;
  return cs.filter(c => c.type !== 'self' && !S.prefs.myc[c.id] && c.createdAt > t).length;
}
function notifyNewComments(prev) {
  let toMe = 0, shared = 0;
  for (const [fid, cs] of Object.entries(S.comments)) {
    const mineFile = !!S.prefs.mine[fid];
    for (const [cid, c] of Object.entries(cs || {})) {
      if (prev[fid]?.[cid] || S.prefs.myc[cid]) continue;
      if (c.type === 'author' && mineFile) toMe++; else if (c.type === 'all') shared++;
    }
  }
  if (toMe) toast(`あなたの文書にコメントが届きました（${toMe}件）`);
  else if (shared) toast(`新しいコメントがあります（${shared}件）`);
  V.panes?.forEach(p => p.updateCBadge());
}

const REPLACED_MSG = 'このファイルは置き換えられました。この画面を閉じると、新しいバージョンに変わります。メモ等は消えるので、必要に応じてスクリーンショットを撮ってください';
function checkPanesForChanges() {
  if (!V.el || V.el.hidden) return;
  let changed = false;
  for (const p of V.panes) {
    const nf = S.files.find(f => f.id === p.id);
    if (!nf) { if (p.noticeState !== 'deleted') { p.noticeState = 'deleted'; p.showNotice('このファイルは削除されました。この画面を閉じると、テーブルからも消えます。'); changed = true; } }
    else if (verOf(nf) !== p.version && p.noticeState !== 'v' + verOf(nf)) { p.noticeState = 'v' + verOf(nf); p.showNotice(REPLACED_MSG); changed = true; }
  }
  if (changed) toast('読んでいるファイルが置き換え・削除されました');
}

async function replaceFile(f) {
  if (!S.prefs.mine[f.id]) return toast('自分がアップしたファイルだけ置き換えられます');
  if (S.table.locked && !S.teacher) return toast('現在アップロードは締め切られています');
  const inp = h('input', { type: 'file', accept: ACCEPT, hidden: true });
  document.body.append(inp);
  const file = await new Promise(res => { inp.onchange = () => res(inp.files[0] || null); inp.click(); });
  inp.remove();
  if (!file) return;
  if (!okFile(file)) return toast('PDF・Word(.docx)・画像のみ使えます');
  const kind = isImage(file) ? '・画像はPDFに変換' : isDocx(file) ? '・WordはPDFに変換' : '';
  const titleIn = h('input', { class: 'input', value: f.title });
  const stat = h('span', {}, ''), bar = h('i');
  let busy = false, ov = null;
  await modal({ title: 'ファイルを置き換える', wide: true, bind: (_c, o) => { ov = o; }, body: [
    h('div', { class: 'cmp' },
      h('div', {}, h('small', {}, 'いまのファイル'), h('b', {}, fileLabel(f))),
      h('div', {}, h('small', {}, '新しいファイル'), h('b', {}, file.name), h('small', { style: { marginTop: '4px' } }, fmtSize(file.size) + kind))),
    h('div', { class: 'field' }, h('label', {}, 'タイトル'), titleIn),
    h('p', { class: 'hint' }, 'テーブル上の位置（アップ順）と署名はそのままです。いま読んでいる人の画面には置き換えのお知らせが出ます。このファイルへの書き込みと「自分用」コメントは、置き換え後は消えます（全員分）。「書き手に伝える」「みんなに共有」のコメントは「旧版」の印が付いて残ります。'),
    h('div', { class: 'stat', style: { display: 'flex', gap: '8px', alignItems: 'center', fontSize: '13px', color: 'var(--muted)' } }, stat, h('span', { class: 'bar' }, bar))
  ], actions: [
    { text: 'キャンセル', value: null, onClick: () => busy ? false : undefined },
    { text: '置き換える', kind: 'primary', onClick: async () => {
      if (busy) return false;
      busy = true; ov.dataset.busy = '1'; ov.querySelectorAll('.modal-f .btn, input').forEach(b => b.disabled = true);
      try {
        const { bytes, pages, thumb, srcType } = await preparePdf(file, t => { stat.textContent = t; });
        const cur = S.files.find(x => x.id === f.id);
        if (!cur) throw new Error('元のファイルが削除されています');
        const version = verOf(cur) + 1, oldKey = blobKeyOf(cur), newKey = f.id + '~v' + version;
        stat.textContent = 'アップロード中…';
        const patch = { title: titleIn.value.trim() || cur.title, srcType, pages, size: bytes.length, thumb, version, blobKey: newKey, replacedAt: Date.now() };
        await S.backend.replaceFile(S.code, f.id, oldKey, newKey, patch, bytesToChunks(bytes), p => { bar.style.width = Math.round(p * 100) + '%'; });
        idb.set('pdf', pdfKey(newKey), bytes.buffer.slice(0)); idb.del('pdf', pdfKey(oldKey));
        S.prefs.seen[f.id] = version; savePrefs();
        toast('置き換えました'); return 'done';
      } catch (e) {
        console.error(e); stat.textContent = '失敗：' + (e.message || e); stat.style.color = 'var(--danger)';
        busy = false; delete ov.dataset.busy; ov.querySelectorAll('.modal-f .btn, input').forEach(b => b.disabled = false);
        return false;
      }
    } }
  ] });
}

  async postComment() {

  drawComments() {

  }
  async deleteComment(c) {
    if (!(await confirmBox('コメントを削除', 'このコメントを削除しますか？', '削除', true))) return;
    if (c.type === 'self') { const n = S.prefs.notes[this.id]; if (n) n.list = n.list.filter(x => x.id !== c.id); savePrefs(); }
    else { await S.backend.deleteComment(S.code, this.id, c.id); delete S.prefs.myc[c.id]; savePrefs(); }
    this.drawComments();
  }
  goPage(n) { const p = this.pages[n - 1]; if (p) this.scroll.scrollTop = p.el.offsetTop - 8; }
  updateCBadge() {
    const f = S.files.find(x => x.id === this.id) || this.file;
    const un = (this.cpanel && !this.cpanel.hidden) ? 0 : unreadCount(f, visibleComments(f));
    this.cbadge.hidden = !un; this.cbadge.textContent = un;
  }
  async load() {
    try {
      const [bytes, ann0] = await Promise.all([getPdfBytes(this.file), idb.get('ann', annKey(this.id))]);
      if (this.dead) return;
      // 書き込み・メモはバージョンごと。置き換えられた後は古いものを捨てる
      let ann = ann0;
      if (ann && (ann.v || 1) !== this.version) { ann = null; idb.del('ann', annKey(this.id)); if (S.prefs.inked[this.id]) { delete S.prefs.inked[this.id]; savePrefs(); } }
      if (ann) this.ann = { pages: ann.pages || {}, memo: '' };
      this.pdf = await openPdf(bytes);
      if (this.dead) return this.pdf.destroy();
      this.io = new IntersectionObserver(es => {
        for (const e of es) {
          const p = e.target._p;
          if (e.isIntersecting) { this.visible.add(p); if (!p.rendered) this.render(p); }
          else { this.visible.delete(p); this.release(p); }
        }
      }, { root: this.scroll, rootMargin: '900px 0px' });
      const els = [];
      for (let i = 1; i <= this.pdf.numPages; i++) {
        const page = await this.pdf.getPage(i); if (this.dead) return;
        const vp = page.getViewport({ scale: 1 });
        const p = { i, page, w: vp.width, h: vp.height, rendered: false, task: null };
        p.pdfC = h('canvas'); p.inkC = h('canvas', { class: 'ink' });
        p.el = h('div', { class: 'page' }, p.pdfC, p.inkC, h('span', { class: 'pnum' }, i));
        p.el._p = p; this.bindInk(p); this.pages.push(p); els.push(p.el);
      }
      this.pagesEl.replaceChildren(...els);
      this.layout(); this.pages.forEach(p => this.io.observe(p.el)); this.onScroll();
    } catch (e) {
      console.error(e);
      this.pagesEl.replaceChildren(h('div', { class: 'pstatus' }, '開けませんでした：' + (e.message || e)));
    }
  }
  layout() {
    if (!this.pages.length) return;
    const ratio = this.scroll.scrollHeight ? this.scroll.scrollTop / this.scroll.scrollHeight : 0;
    const avail = Math.max(200, Math.min(this.scroll.clientWidth - 24, 1000));
    for (const p of this.pages) {
      p.cssW = Math.round(avail * this.zoom); p.cssH = Math.round(p.cssW * p.h / p.w);
      p.el.style.width = p.cssW + 'px'; p.el.style.height = p.cssH + 'px'; p.rendered = false;
    }
    this.scroll.scrollTop = ratio * this.scroll.scrollHeight;
    for (const p of this.visible) this.render(p);
  }
  setZoom(d) {
    const i = ZOOMS.indexOf(this.zoom); const ni = Math.max(0, Math.min(ZOOMS.length - 1, (i < 0 ? 3 : i) + d));
    this.zoom = ZOOMS[ni]; this.layout(); toast(Math.round(this.zoom * 100) + '%');
  }
  async render(p) {
    p.rendered = true;
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    const vp = p.page.getViewport({ scale: p.cssW / p.w * dpr });
    if (p.task) { p.task.cancel(); p.task = null; }
    const off = document.createElement('canvas'); off.width = Math.floor(vp.width); off.height = Math.floor(vp.height);
    const task = p.task = p.page.render({ canvasContext: off.getContext('2d'), viewport: vp });
    try { await task.promise; } catch { return; }
    if (p.task !== task || this.dead) return;
    p.task = null;
    p.pdfC.width = off.width; p.pdfC.height = off.height; p.pdfC.getContext('2d').drawImage(off, 0, 0);
    p.inkC.width = off.width; p.inkC.height = off.height; this.drawInk(p);
  }
  release(p) {
    if (p.task) { p.task.cancel(); p.task = null; }
    p.pdfC.width = p.pdfC.height = 0; p.inkC.width = p.inkC.height = 0; p.rendered = false;
  }
  onScroll() {
    if (this.pages.length) {
      const y = this.scroll.scrollTop + this.scroll.clientHeight / 3;
      let cur = 1; for (const p of this.pages) { if (p.el.offsetTop <= y) cur = p.i; else break; }
      this.curPage = cur; this.pgEl.textContent = `${cur} / ${this.pages.length}`;
    }
    if (V.sync && !V.syncing && V.panes.length > 1) {
      const s = this.scroll, r = s.scrollTop / Math.max(1, s.scrollHeight - s.clientHeight);
      V.syncing = true;
      for (const o of V.panes) if (o !== this) o.scroll.scrollTop = r * (o.scroll.scrollHeight - o.scroll.clientHeight);
      requestAnimationFrame(() => requestAnimationFrame(() => { V.syncing = false; }));
    }
  }
  /* ---- ink ---- */
  drawInk(p, extra) {
    const c = p.inkC; if (!c.width) return;
    const ctx = c.getContext('2d'); ctx.clearRect(0, 0, c.width, c.height);
    const list = (this.ann.pages[p.i] || []).concat(extra ? [extra] : []);
    for (const s of list) drawStroke(ctx, s, c.width, c.height);
  }
  norm(e, p) {
    const r = p.inkC.getBoundingClientRect();
    return [Math.round((e.clientX - r.left) / r.width * 1e4) / 1e4, Math.round((e.clientY - r.top) / r.height * 1e4) / 1e4];
  }
  bindInk(p) {
    const c = p.inkC;
    c.addEventListener('pointerdown', e => {
      setActive(this);
      if (V.tool === 'hand' || (e.pointerType === 'mouse' && e.button !== 0)) return;
      e.preventDefault(); c.setPointerCapture(e.pointerId);
      const [x, y] = this.norm(e, p);
      if (V.tool === 'eraser') { this.drag = { p, id: e.pointerId, removed: [] }; this.eraseAt(p, x, y); }
      else {
        const t = V.tool;
        this.drag = { p, id: e.pointerId, stroke: { t, c: V.colors[t], w: SIZES[t][V.size[t]] / 800, p: [x, y] } };
        this.drawInk(p, this.drag.stroke);
      }
    });
    c.addEventListener('pointermove', e => {
      const d = this.drag; if (!d || d.p !== p || d.id !== e.pointerId) return;
      e.preventDefault();
      const evs = e.getCoalescedEvents ? e.getCoalescedEvents() : [e];
      for (const ev of (evs.length ? evs : [e])) {
        const [x, y] = this.norm(ev, p);
        if (d.stroke) { const a = d.stroke.p; const lx = a[a.length - 2], ly = a[a.length - 1]; if (Math.abs(x - lx) + Math.abs(y - ly) > 0.0015) a.push(x, y); }
        else this.eraseAt(p, x, y);
      }
      if (d.stroke && !this.raf) this.raf = requestAnimationFrame(() => { this.raf = 0; if (this.drag?.stroke) this.drawInk(p, this.drag.stroke); });
    });
    const end = e => {
      const d = this.drag; if (!d || d.p !== p || d.id !== e.pointerId) return;
      this.drag = null;
      if (d.stroke) {
        (this.ann.pages[p.i] ||= []).push(d.stroke);
        this.hist.push({ op: 'add', page: p.i, stroke: d.stroke }); this.drawInk(p); this.save();
      } else if (d.removed.length) { this.hist.push({ op: 'del', page: p.i, items: d.removed }); this.save(); }
    };
    c.addEventListener('pointerup', end); c.addEventListener('pointercancel', end);
  }
  eraseAt(p, x, y) {
    const arr = this.ann.pages[p.i]; if (!arr || !arr.length) return;
    const ar = p.cssH / p.cssW, r = 12 / p.cssW; let hit = false;
    for (let k = arr.length - 1; k >= 0; k--) {
      const s = arr[k];
      if (strokeHit(s, x, y, r + s.w / 2, ar)) { this.drag.removed.push({ s, idx: k }); arr.splice(k, 1); hit = true; }
    }
    if (hit) this.drawInk(p);
  }
  undoOnce() {
    const a = this.hist.pop(); if (!a) return toast('元に戻す操作はありません');
    if (a.op === 'add') { const arr = this.ann.pages[a.page] || []; const i = arr.indexOf(a.stroke); if (i >= 0) arr.splice(i, 1); }
    else if (a.op === 'del') { const arr = (this.ann.pages[a.page] ||= []); for (const it of a.items.slice().reverse()) arr.splice(Math.min(it.idx, arr.length), 0, it.s); }
    else if (a.op === 'clear') this.ann.pages = a.pages;
    this.pages.forEach(p => this.drawInk(p)); this.save();
  }
  async clearAll() {
    if (!Object.values(this.ann.pages).some(a => a.length)) return toast('書き込みはありません');
    if (!(await confirmBox('書き込みを消す', `「${fileLabel(liveFile(this.file))}」の線・マーカーをすべて消します（コメントは残ります）。`, '消す', true))) return;
    this.hist.push({ op: 'clear', pages: this.ann.pages }); this.ann.pages = {};
    this.pages.forEach(p => this.drawInk(p)); this.save();
  }
  save(now) {
    clearTimeout(this.st);
    const run = () => {
      this.st = null;
      const has = Object.values(this.ann.pages).some(a => a.length);
      // 置き換え後の新版に古い書き込みが混ざらないよう、読み込んだ版の番号と一緒に保存
      idb.get('ann', annKey(this.id)).then(cur => {
        if (cur && (cur.v || 1) > this.version) return;
        idb.set('ann', annKey(this.id), { v: this.version, pages: this.ann.pages, updatedAt: Date.now() });
      });
      const want = has ? this.version : undefined;
      if (S.prefs.inked[this.id] !== want) { has ? (S.prefs.inked[this.id] = this.version) : delete S.prefs.inked[this.id]; savePrefs(); }
    };
    if (now) run(); else this.st = setTimeout(run, 400);
  }
  showNotice(msg) {
    this.noticeEl.replaceChildren(icon('alert'), h('div', {}, msg));
    this.noticeEl.hidden = false;
  }
  destroy() {
    if (this.st) this.save(true);
    this.dead = true; this.io?.disconnect();
    this.pages.forEach(p => p.task?.cancel());
    this.pdf?.destroy();
  }
}
function drawStroke(ctx, s, W, H) {
  const a = s.p; if (!a || a.length < 2) return;
  ctx.save();
  ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.strokeStyle = s.c; ctx.lineWidth = Math.max(1, s.w * W);
  if (s.t === 'marker') ctx.globalAlpha = 0.4;
  ctx.beginPath(); ctx.moveTo(a[0] * W, a[1] * H);
  if (a.length === 2) ctx.lineTo(a[0] * W + 0.1, a[1] * H);
  else {
    for (let i = 2; i < a.length - 2; i += 2) {
      const mx = (a[i] + a[i + 2]) / 2, my = (a[i + 1] + a[i + 3]) / 2;
      ctx.quadraticCurveTo(a[i] * W, a[i + 1] * H, mx * W, my * H);
    }
    ctx.lineTo(a[a.length - 2] * W, a[a.length - 1] * H);
  }
  ctx.stroke(); ctx.restore();
}
function strokeHit(s, x, y, r, ar) {
  const a = s.p, r2 = r * r;
  const d2 = (px, py) => { const dx = px - x, dy = (py - y) * ar; return dx * dx + dy * dy; };
  if (a.length === 2) return d2(a[0], a[1]) <= r2;
  for (let i = 0; i < a.length - 2; i += 2) {
    const x1 = a[i], y1 = a[i + 1] * ar, x2 = a[i + 2], y2 = a[i + 3] * ar, py = y * ar;
    const vx = x2 - x1, vy = y2 - y1, L = vx * vx + vy * vy;
    let t = L ? ((x - x1) * vx + (py - y1) * vy) / L : 0; t = Math.max(0, Math.min(1, t));
    const dx = x1 + t * vx - x, dy = y1 + t * vy - py;
    if (dx * dx + dy * dy <= r2) return true;
  }
  return false;
}
